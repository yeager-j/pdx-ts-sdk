namespace = pdx_hardening

country_event = {
	id = pdx_hardening.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				num_owned_planets >= 1
			}
			log = PDX_HARDENING_ENTRY
			set_country_flag = pdx_hardening_entry_fired
			every_owned_planet = {
				save_event_target_as = pdx_hardening_chain_planet
			}
			event_target:pdx_hardening_chain_planet = {
				log = PDX_HARDENING_TARGET_AVAILABLE_IN_ENTRY
			}
			country_event = {
				id = pdx_hardening.2
				days = 1
				scopes = {
					from = event_target:pdx_hardening_chain_planet
				}
			}
			country_event = {
				id = pdx_hardening.3
				days = 1
			}
		}
	}
}

country_event = {
	id = pdx_hardening.2

	is_triggered_only = yes

		}
	}
}
