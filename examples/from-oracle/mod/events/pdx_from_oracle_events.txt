namespace = pdx_from_oracle

country_event = {
	id = pdx_from_oracle.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		log = "PDX_FROM_ORACLE_ENTRY"
		if = {
			limit = { is_scope_type = country }
			log = "PDX_FROM_ORACLE_TOP_THIS_COUNTRY"
		}
		root = {
			if = {
				limit = { is_scope_type = country }
				log = "PDX_FROM_ORACLE_TOP_ROOT_COUNTRY"
			}
		}

		country_event = { id = pdx_from_oracle.2 }

		every_owned_planet = {
			limit = { is_capital = yes }
			if = {
				limit = { is_scope_type = colony }
				log = "PDX_FROM_ORACLE_ITERATOR_THIS_COLONY"
			}
			root = {
				if = {
					limit = { is_scope_type = country }
					log = "PDX_FROM_ORACLE_ITERATOR_ROOT_COUNTRY"
				}
			}
			colony_event = { id = pdx_from_oracle.3 }
		}

		capital_scope.solar_system = {
			spawn_system = {
				min_distance = 10
				max_distance = 30
				min_jumps = 1
				max_jumps = 2
				hyperlane = no
				initializer = pdx_from_oracle_split_root
			}
		}
	}
}

country_event = {
	id = pdx_from_oracle.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		from = {
			if = {
				limit = { is_scope_type = country }
				log = "PDX_FROM_ORACLE_TOP_FROM_COUNTRY"
			}
			if = {
				limit = { is_scope_type = planet }
				log = "PDX_FROM_ORACLE_TOP_FROM_PLANET"
			}
		}
	}
}

colony_event = {
	id = pdx_from_oracle.3
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		from = {
			if = {
				limit = { is_scope_type = country }
				log = "PDX_FROM_ORACLE_ITERATOR_FROM_COUNTRY"
			}
			if = {
				limit = { is_scope_type = colony }
				log = "PDX_FROM_ORACLE_ITERATOR_FROM_COLONY"
			}
		}
	}
}

planet_event = {
	id = pdx_from_oracle.4
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		from = {
			if = {
				limit = { is_scope_type = country }
				log = "PDX_FROM_ORACLE_SPLIT_FROM_COUNTRY"
			}
			if = {
				limit = { is_scope_type = planet }
				log = "PDX_FROM_ORACLE_SPLIT_FROM_PLANET"
			}
		}
	}
}
