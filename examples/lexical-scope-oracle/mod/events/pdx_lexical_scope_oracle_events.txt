namespace = pdx_lexical_scope_oracle

country_event = {
	id = pdx_lexical_scope_oracle.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		log = "PDX_LEXICAL_SCOPE_ORACLE_ENTRY"
		if = {
			limit = { is_scope_type = country }
			log = "PDX_LEXICAL_SCOPE_ORACLE_IF_SAME_COUNTRY"
		}
		hidden_effect = {
			if = {
				limit = { is_scope_type = country }
				log = "PDX_LEXICAL_SCOPE_ORACLE_HIDDEN_SAME_COUNTRY"
			}
		}
		every_owned_planet = {
			limit = { is_capital = yes }
			prev = {
				if = {
					limit = { is_scope_type = country }
					log = "PDX_LEXICAL_SCOPE_ORACLE_PREV_COUNTRY"
				}
			}
			owner = {
				prevprev = {
					if = {
						limit = { is_scope_type = country }
						log = "PDX_LEXICAL_SCOPE_ORACLE_PREVPREV_COUNTRY"
					}
				}
			}
		}
	}
}
