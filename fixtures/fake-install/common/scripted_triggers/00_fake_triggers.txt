# Fixture scripted triggers. One case per parameter rule the reader has to get
# right; the bodies are invented, not copied from the game.

fake_no_params = {
	is_country_type = default
}

fake_defaulted = {
	years_passed > $YEARS|10$
}

fake_required = {
	has_technology = $TECH$
}

fake_conditional = {
	[[FLAG]
		has_country_flag = $NAME$
	]
	is_country_type = default
}

# A name written with a default in one place and without one in another. The
# second substitution has nothing to fall back on, so the parameter is
# required: reading the first occurrence as permission to omit it left a
# substitution site with nothing to substitute.
fake_mixed_default = {
	years_passed > $YEARS|10$
	has_country_flag = $YEARS$
}

# A region that forces nothing. Its only substitution is a parameter the body
# already requires outside it, so activating the region asks the caller for
# nothing new and the signature stays one flat shape. Vanilla's
# `roll_secret_faction_internal` is this.
fake_inert_region = {
	has_technology = $TECH$
	[[TOGGLE]
		has_country_flag = $TECH$
	]
}

# A region beside its negated twin, which is the shape vanilla actually writes:
# `add_random_trait_evopred` is two near-identical branches chosen by whether
# `SPECIES` was supplied. Exactly one always runs, so `$NAME$` is always
# substituted and NAME is required however TOGGLE is answered. Reading the
# negation as if it were `[[TOGGLE]` makes this look like a dependency and
# refuses `{ NAME: "x" }`, which is the ordinary call.
fake_negated_pair = {
	[[TOGGLE]
		has_country_flag = $NAME$
	]
	[[!TOGGLE]
		has_country_flag = $NAME$
	]
}

# A region whose flag the body also substitutes unconditionally. The caller
# must supply GATE, so the region is always active, so what it substitutes is
# required too. The call that omits GATE is one nobody can make.
fake_mandatory_flag = {
	has_country_flag = $GATE$
	[[GATE]
		has_technology = $INNER$
	]
}

# Two cases the scope inference has to widen on rather than narrow, so the
# generator's attribution report is testable without an install.

# A key no rule declares — a mod's, or one a game patch introduced. The
# inference must report it and leave the binding unconstrained.
fake_unknown_key = {
	some_key_no_rule_declares = yes
}

# Two conditions whose scopes are disjoint, so the intersection empties. The
# fallback is unconstrained; the report names it rather than hiding it.
fake_dual_scope = {
	is_country_type = default
	is_planet_class = pc_continental
}
